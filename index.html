Set 1 — Self‑Driving Car CNN Regression

Perform EDA on the self‑driving car dataset and develop a CNN‑based regression model to predict steering angles from road images. Train the model and visualize actual versus predicted steering angles.

Restriction: Use a prepared dataset and a reduced CNN architecture or the NVIDIA architecture provided in the lab.
Single self‑contained script → save as set1.py, run top‑to‑bottom. Needs driving_log.csv + IMG/ folder beside it.
set1.py
# pip install tensorflow opencv-python scikit-learn pandas matplotlib
# Dataset : Udidity self-driving car -> driving_log.csv + IMG/ folder
# Sample  : first 10,000 rows used (random 10k) so it finishes in a 3-hour lab.

import os
import cv2
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, mean_absolute_error
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, Dense, Dropout, Flatten, Lambda

# ---------- 1. LOAD DATA (detect / skip header, cast steering to float) ----------
CSV     = 'driving_log.csv'   # columns: center,left,right,steering,throttle,brake,speed
IMG_DIR = 'IMG'

raw = pd.read_csv(CSV, header=None)
# if the first row is a header (contains the word 'steering'), drop it
if raw.iloc[0].astype(str).str.lower().str.contains('steering').any():
    raw = raw.iloc[1:].reset_index(drop=True)
raw.columns = ['center','left','right','steering','throttle','brake','speed']
raw['steering'] = raw['steering'].astype(float)
raw['speed']    = raw['speed'].astype(float)

# sample for lab time
raw = raw.sample(n=min(10000, len(raw)), random_state=42).reset_index(drop=True)
print("Rows used:", len(raw))
print(raw.head())

# ---------- 2. EDA ----------
plt.figure(figsize=(10,4))
plt.hist(raw['steering'], bins=50, color='steelblue')
plt.xlabel('Steering Angle'); plt.ylabel('Count')
plt.title('Distribution of Steering Angles')
plt.tight_layout(); plt.savefig('steering_dist.png'); plt.show()

plt.figure(figsize=(8,4))
plt.scatter(raw['speed'], raw['steering'], s=1, alpha=0.3)
plt.xlabel('Speed'); plt.ylabel('Steering')
plt.title('Speed vs Steering')
plt.tight_layout(); plt.savefig('speed_vs_steering.png'); plt.show()

fig, axes = plt.subplots(1,3, figsize=(15,5))
for i, col in enumerate(['center','left','right']):
    p = raw.iloc[0][col].strip()
    if not os.path.exists(p):
        p = os.path.join(IMG_DIR, os.path.basename(p))
    axes[i].imshow(plt.imread(p)); axes[i].set_title(col); axes[i].axis('off')
plt.tight_layout(); plt.savefig('sample_cameras.png'); plt.show()

# ---------- 3. PREPROCESS + AUGMENT ----------
IMG_H, IMG_W = 66, 200
BATCH     = 32
CORRECTION = 0.2          # left/right camera steering correction

def resolve(path):
    p = path.strip()
    return p if os.path.exists(p) else os.path.join(IMG_DIR, os.path.basename(p))

def preprocess_image(path):
    img = cv2.imread(resolve(path))
    if img is None:
        raise FileNotFoundError(f"Cannot read image: {path}")
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = img[60:140, :, :]                 # crop sky + hood
    return cv2.resize(img, (IMG_W, IMG_H))

def augment(img, steering):
    if np.random.rand() < 0.5:             # horizontal flip
        img = cv2.flip(img, 1)
        steering = -steering
    if np.random.rand() < 0.3:             # brightness jitter
        b = np.random.uniform(0.6, 1.4)
        img = np.clip(img * b, 0, 255).astype(np.uint8)
    return img, steering

# ---------- 4. GENERATOR (center + left + right with +/-0.2 correction) ----------
samples = raw[['center','left','right','steering']].values
train, val = train_test_split(samples, test_size=0.2, random_state=42)
print("Train:", len(train), " Val:", len(val))

def generator(samples, batch_size=BATCH):
    while True:
        idx = np.random.permutation(len(samples))
        for offset in range(0, len(samples), batch_size):
            batch = samples[idx[offset:offset+batch_size]]
            imgs, angles = [], []
            for center, left, right, steer in batch:
                steer = float(steer)
                for cam, s in ((center, steer),
                               (left,  steer + CORRECTION),
                               (right, steer - CORRECTION)):
                    img = preprocess_image(cam)
                    img, s = augment(img, s)
                    imgs.append(img); angles.append(s)
            yield np.array(imgs, dtype=np.float32), np.array(angles, dtype=np.float32)

# ---------- 5. MODEL (NVIDIA architecture exactly) ----------
model = Sequential([
    Lambda(lambda x: x / 127.5 - 1.0, input_shape=(IMG_H, IMG_W, 3)),
    Conv2D(24, (5,5), strides=(2,2), activation='elu'),
    Conv2D(36, (5,5), strides=(2,2), activation='elu'),
    Conv2D(48, (5,5), strides=(2,2), activation='elu'),
    Conv2D(64, (3,3), activation='elu'),
    Conv2D(64, (3,3), activation='elu'),
    Flatten(),
    Dense(100, activation='elu'),
    Dropout(0.5),
    Dense(50,  activation='elu'),
    Dense(10,  activation='elu'),
    Dense(1)                                   # single steering output
])
model.compile(optimizer='adam', loss='mse', metrics=['mae'])
model.summary()
# input (None, 66, 200, 3)  ->  output (None, 1)

# ---------- 6. TRAIN ----------
EPOCHS = 10
history = model.fit(
    generator(train), steps_per_epoch=len(train)//BATCH,
    epochs=EPOCHS,
    validation_data=generator(val), validation_steps=len(val)//BATCH,
    verbose=1
)
model.save('steering_model.h5')

# ---------- 7. EVALUATE + VISUALIZE (first 1000 val samples only) ----------
N_EVAL = 1000
eval_batch = val[:N_EVAL]
eval_imgs  = np.array([preprocess_image(s[0]) for s in eval_batch], dtype=np.float32)
eval_steer = np.array([float(s[3]) for s in eval_batch], dtype=np.float32)
preds = model.predict(eval_imgs).flatten()      # (1000,)

mse = mean_squared_error(eval_steer, preds)
mae = mean_absolute_error(eval_steer, preds)
print(f"MSE : {mse:.6f}")
print(f"MAE : {mae:.6f}")

plt.figure(figsize=(14,4))
plt.plot(eval_steer, label='Actual',    alpha=0.8)
plt.plot(preds,      label='Predicted', alpha=0.8)
plt.xlabel('Sample'); plt.ylabel('Steering Angle')
plt.title('Actual vs Predicted Steering Angles')
plt.legend(); plt.tight_layout()
plt.savefig('actual_vs_predicted.png'); plt.show()

plt.figure(figsize=(6,6))
plt.scatter(eval_steer, preds, s=4, alpha=0.3)
lo = min(eval_steer.min(), preds.min()) - 0.05
hi = max(eval_steer.max(), preds.max()) + 0.05
plt.plot([lo,hi],[lo,hi],'r--')
plt.xlabel('Actual'); plt.ylabel('Predicted')
plt.title('Scatter: Actual vs Predicted')
plt.tight_layout(); plt.savefig('scatter.png'); plt.show()

Set 3 — Quora Question Pair Similarity NLP / TF‑IDF

Perform text preprocessing and feature extraction using TF‑IDF on the Quora Question Pair dataset. Implement Logistic Regression or Linear SVM to classify question pairs as duplicate or non‑duplicate. Evaluate using accuracy and F1‑score.

Single self‑contained script → save as set3.py. Needs quora_question_pairs.csv. Sampled to 30,000 rows — no N×N cosine matrix, so it will not OOM.
set3.py
# pip install pandas numpy scikit-learn scipy matplotlib seaborn nltk
# Dataset : Quora Question Pairs -> quora_question_pairs.csv
# Sample  : 30,000 rows (random, random_state=42) so it finishes in a 3-hour lab.

import re
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import nltk
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer
from scipy.sparse import hstack
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.svm import LinearSVC
from sklearn.metrics import (accuracy_score, f1_score, confusion_matrix,
                             classification_report)

nltk.download('stopwords', quiet=True)
nltk.download('wordnet',  quiet=True)

# ---------- 1. LOAD + SAMPLE ----------
df = pd.read_csv('quora_question_pairs.csv')
print("Full dataset:", df.shape)

if len(df) > 50000:                       # sample for lab time
    df = df.sample(30000, random_state=42).reset_index(drop=True)
    print("Sampled to 30,000 rows for the 3-hour lab.")
print("Working set:", df.shape)

# ---------- 2. TEXT PREPROCESSING ----------
stop_words  = set(stopwords.words('english'))
lemmatizer  = WordNetLemmatizer()

def clean(text):
    if not isinstance(text, str):
        return ""
    text = text.lower()
    text = re.sub(r'[^a-z\s]', '', text)
    tokens = [lemmatizer.lemmatize(t) for t in text.split() if t not in stop_words]
    return ' '.join(tokens)

df['q1_clean'] = df['question1'].apply(clean)
df['q2_clean'] = df['question2'].apply(clean)
print(df[['question1','q1_clean']].head())

# ---------- 3. TRAIN/TEST SPLIT FIRST (prevent TF-IDF leakage) ----------
X_df, y = df[['q1_clean','q2_clean']], df['is_duplicate']
X_train_df, X_test_df, y_train, y_test = train_test_split(
    X_df, y, test_size=0.2, random_state=42, stratify=y)

# ---------- 4. TF-IDF fit ONLY on training questions ----------
tfidf = TfidfVectorizer(max_features=20000, ngram_range=(1,2), sublinear_tf=True)
tfidf.fit(X_train_df['q1_clean'] + ' ' + X_train_df['q2_clean'])

q1_tr = tfidf.transform(X_train_df['q1_clean'])
q2_tr = tfidf.transform(X_train_df['q2_clean'])
q1_te = tfidf.transform(X_test_df['q1_clean'])
q2_te = tfidf.transform(X_test_df['q2_clean'])

# row-wise cosine similarity (sparse -> no N x N dense matrix, no OOM)
cos_tr = np.asarray(q1_tr.multiply(q2_tr).sum(axis=1)).ravel().reshape(-1,1)
cos_te = np.asarray(q1_te.multiply(q2_te).sum(axis=1)).ravel().reshape(-1,1)

X_train = hstack([q1_tr, q2_tr, cos_tr])    # (24000, 20001)
X_test  = hstack([q1_te, q2_te, cos_te])    # (6000,  20001)
print("Train matrix:", X_train.shape, " Test matrix:", X_test.shape)

# ---------- 5. LOGISTIC REGRESSION (main) ----------
lr = LogisticRegression(max_iter=1000, C=1.0, n_jobs=-1)
lr.fit(X_train, y_train)
y_pred_lr = lr.predict(X_test)

# ---------- 6. LINEAR SVM (comparison) ----------
svm = LinearSVC(C=0.5, max_iter=2000, random_state=42)
svm.fit(X_train, y_train)
y_pred_svm = svm.predict(X_test)

# ---------- 7. EVALUATION ----------
for name, preds in [('Logistic Regression', y_pred_lr), ('Linear SVM', y_pred_svm)]:
    print(f"\n=== {name} ===")
    print(f"Accuracy : {accuracy_score(y_test, preds):.4f}")
    print(f"F1 Score : {f1_score(y_test, preds):.4f}")
    print(classification_report(y_test, preds, target_names=['Non-Duplicate','Duplicate']))

fig, axes = plt.subplots(1,2, figsize=(12,5))
for ax, preds, title in zip(axes, [y_pred_lr, y_pred_svm], ['Logistic Regression','Linear SVM']):
    sns.heatmap(confusion_matrix(y_test, preds), annot=True, fmt='d', cmap='Greens', ax=ax,
                xticklabels=['Non-Dup','Dup'], yticklabels=['Non-Dup','Dup'])
    ax.set_title(f'{title} - Acc {accuracy_score(y_test,preds):.3f}')
    ax.set_xlabel('Predicted'); ax.set_ylabel('Actual')
plt.tight_layout(); plt.savefig('quora_confusion.png'); plt.show()

Set 4 — Music Generation Char‑level RNN

Develop a character‑level RNN model using ABC Notation for music generation. Perform character encoding, prepare input sequences, train the model and generate a short music sequence. Convert the generated ABC Notation into MIDI format.

Restriction: Use a prepared dataset and a simplified RNN model. Pretrained weights or a previously trained model should not replace the required implementation.
Single self‑contained script → save as set4.py. Needs abc_dataset.txt. Corpus capped to first 50,000 chars so training (20 epochs) fits in lab time. No pretrained weights.
set4.py
# pip install tensorflow numpy music21 mido
# Dataset : prepared ABC notation corpus -> abc_dataset.txt
# Sample  : first 50,000 characters only, so 20 epochs finish in a 3-hour lab.

import os
import numpy as np
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, LSTM, Dense, Dropout

# ---------- 1. LOAD + CHARACTER ENCODE ----------
with open('abc_dataset.txt', 'r', encoding='utf-8') as f:
    text = f.read()

MAX_CHARS = 50000
if len(text) > MAX_CHARS:
    text = text[:MAX_CHARS]
    print(f"Corpus capped to first {MAX_CHARS} characters for the 3-hour lab.")
print("Corpus length:", len(text))

chars     = sorted(set(text))
char2idx  = {c:i for i,c in enumerate(chars)}
idx2char  = {i:c for c,i in char2idx.items()}
vocab_size = len(chars)
print("Vocabulary:", vocab_size, "characters")

data = np.array([char2idx[c] for c in text], dtype=np.int32)   # (len(text),)

# ---------- 2. PREPARE INPUT SEQUENCES ----------
CHUNK = 40
X, y_seq = [], []
for i in range(0, len(data) - CHUNK - 1, CHUNK):
    X.append(data[i:i+CHUNK])               # 40 chars in
    y_seq.append(data[i+1:i+CHUNK+1])       # next-char targets
X = np.array(X); y_seq = np.array(y_seq)
print("Sequences:", X.shape, y_seq.shape)
# X: (n_seq, 40)   y_seq: (n_seq, 40)

# ---------- 3. MODEL ----------
# LSTM is the RNN variant used here; 2 x LSTM(128) is the simplified
# model required by the restriction. No pretrained weights are loaded.
EMBED_DIM  = 64
LSTM_UNITS = 128

model = Sequential([
    Embedding(vocab_size, EMBED_DIM, input_length=CHUNK),
    LSTM(LSTM_UNITS, return_sequences=True),
    Dropout(0.2),
    LSTM(LSTM_UNITS, return_sequences=True),
    Dropout(0.2),
    Dense(vocab_size, activation='softmax')
])
model.compile(loss='sparse_categorical_crossentropy', optimizer='adam')
model.summary()
# input (None, 40)  ->  output (None, 40, vocab_size)

# ---------- 4. TRAIN ----------
EPOCHS = 20
history = model.fit(X, y_seq, batch_size=64, epochs=EPOCHS,
                    validation_split=0.1, verbose=1)

# ---------- 5. GENERATE ----------
def generate_text(model, seed, length=300, temperature=0.8):
    generated = seed
    for _ in range(length):
        x = np.array([[char2idx[c] for c in generated[-CHUNK:]]])   # (1, 40)
        # [0, -1, :] takes the LAST timestep prediction because the LSTM
        # uses return_sequences=True -> output shape is (1, CHUNK, vocab_size).
        probs = model.predict(x, verbose=0)[0, -1, :].astype(np.float64)
        logits = np.log(probs + 1e-8) / temperature
        probs  = np.exp(logits)
        probs  = probs / probs.sum()
        next_idx = np.random.choice(len(probs), p=probs)
        generated += idx2char[next_idx]
    return generated

seed = text[:CHUNK]
generated = generate_text(model, seed, length=300, temperature=0.8)
print("Raw generated text:\n", generated[:300])

# ---------- 6. VALIDATE / CLEAN ABC ----------
HEADER = "X:1\nT:Generated Tune\nM:4/4\nL:1/8\nK:C\n"
if not generated.lstrip().startswith("X:"):
    generated = HEADER + generated
    print("Prepended ABC header (text did not start with 'X:').")

VALID = set("XTLQK:|/[]()abcdefghijklmnopqrstuvwxyzABCDEF"
            "0123456789^_=\n \t-+")
generated = ''.join(ch for ch in generated if ch in VALID)

with open('generated.abc', 'w', encoding='utf-8') as f:
    f.write(generated)
print("Saved generated.abc")

# ---------- 7. ABC -> MIDI (music21, mido fallback) ----------
try:
    from music21 import converter
    score = converter.parse("generated.abc", format="abc")
    score.write('midi', fp='generated.mid')
    print("MIDI saved via music21.")
except Exception as e:
    print("music21 failed:", e)
    print("Falling back to mido manual conversion.")
    from mido import Message, MidiFile, MidiTrack
    midi  = MidiFile()
    track = MidiTrack()
    midi.tracks.append(track)
    track.append(Message('program_change', program=0, time=0))
    NOTE = {'C':0,'D':2,'E':4,'F':5,'G':7,'A':9,'B':11}
    for ch in generated:
        if ch in NOTE:
            pitch = 60 + NOTE[ch]
            track.append(Message('note_on',  note=pitch, velocity=64, time=0))
            track.append(Message('note_off', note=pitch, velocity=64, time=160))
    midi.save('generated.mid')
    print("MIDI saved via mido.")

if os.path.exists('generated.mid'):
    print(f"generated.mid exists, size = {os.path.getsize('generated.mid')} bytes")
else:
    print("generated.mid NOT created")
