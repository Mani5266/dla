from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, Dense, Dropout, Flatten, Lambda, Cropping2D

def build_nvidia_model():
    model = Sequential([
        # input 66x200x3 (preprocessed)
        Lambda(lambda x: x / 127.5 - 1.0, input_shape=(66,200,3)),
        Conv2D(24, (5,5), strides=(2,2), activation='elu'),
        Conv2D(36, (5,5), strides=(2,2), activation='elu'),
        Conv2D(48, (5,5), strides=(2,2), activation='elu'),
        Conv2D(64, (3,3), activation='elu'),
        Conv2D(64, (3,3), activation='elu'),
        Flatten(),
        Dense(100, activation='elu'),
        Dropout(0.5),
        Dense(50, activation='elu'),
        Dense(10, activation='elu'),
        Dense(1)                       # single steering output
    ])
    model.compile(optimizer='adam', loss='mse', metrics=['mae'])
    return model

Data Generator & Training
train.py
import cv2, numpy as np
from sklearn.model_selection import train_test_split

IMG_H, IMG_W = 66, 200
BATCH = 32

def preprocess_image(path):
    img = cv2.imread(path.strip())
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = img[60:140, :, :]                    # crop sky/hood
    img = cv2.resize(img, (IMG_W, IMG_H))
    return img

def augment(img, steering):
    if np.random.rand() < 0.5:
        img = cv2.flip(img, 1)
        steering = -steering
    if np.random.rand() < 0.3:
        brightness = np.random.uniform(0.6, 1.4)
        img = np.clip(img * brightness, 0, 255).astype(np.uint8)
    return img, steering

def generator(samples, batch_size=BATCH):
    while True:
        np.random.shuffle(samples)
        for offset in range(0, len(samples), batch_size):
            batch = samples[offset:offset+batch_size]
            imgs, angles = [], []
            for center, left, right, steer in batch:
                img = preprocess_image(center)
                img, steer = augment(img, steer)
                imgs.append(img); angles.append(steer)
            yield np.array(imgs), np.array(angles)

# build sample list from CSV
samples = list(df[['center','left','right','steering']].values)
train, val = train_test_split(samples, test_size=0.2, random_state=42)

model = build_nvidia_model()
history = model.fit(
    generator(train), steps_per_epoch=len(train)//BATCH, epochs=15,
    validation_data=generator(val), validation_steps=len(val)//BATCH,
    verbose=1
)
model.save('steering_model.h5')

import pandas as pd, numpy as np, re, string, nltk
from nltk.corpus import stopwords
from nltk.stem import WordNetLemmatizer
nltk.download('stopwords', quiet=True); nltk.download('wordnet', quiet=True)

df = pd.read_csv('quora_question_pairs.csv')   # columns: id, qid1, qid2, question1, question2, is_duplicate
print("Shape:", df.shape)
print(df['is_duplicate'].value_counts(normalize=True))

stop_words = set(stopwords.words('english'))
lemmatizer = WordNetLemmatizer()

def clean(text):
    if not isinstance(text, str): return ""
    text = text.lower()
    text = re.sub(r'[^a-z\s]', '', text)            # remove digits / punctuation
    tokens = text.split()
    tokens = [lemmatizer.lemmatize(t) for t in tokens if t not in stop_words]
    return ' '.join(tokens)

df['q1_clean'] = df['question1'].apply(clean)
df['q2_clean'] = df['question2'].apply(clean)
print(df[['question1','q1_clean']].head())

Develop a character‑level RNN model using ABC Notation for music generation. Perform character encoding, prepare input sequences, train the model and generate a short music sequence. Convert the generated ABC Notation into MIDI format.

Character Encoding & Sequence Preparation
data.py
import numpy as np

# load ABC notation corpus
with open('abc_dataset.txt', 'r', encoding='utf-8') as f:
    text = f.read()

print("Corpus length:", len(text))

# character‑level encoding
chars = sorted(set(text))
char2idx = {c:i for i,c in enumerate(chars)}
idx2char = {i:c for c,i in char2idx.items()}
vocab_size = len(chars)
print("Vocabulary:", vocab_size, "characters")

# encode entire corpus
data = np.array([char2idx[c] for c in text], dtype=np.int32)

# build input sequences (chunk length = 40)
CHUNK = 40
X, y_seq = [], []
for i in range(0, len(data) - CHUNK - 1, CHUNK):
    X.append(data[i:i+CHUNK])
    y_seq.append(data[i+1:i+CHUNK+1])          # next‑char target

X = np.array(X); y_seq = np.array(y_seq)
print("Sequences:", X.shape, y_seq.shape)
Character‑level RNN Model

from tensorflow.keras.models import Sequential

Generate Music from Model
generate.py
import numpy as np

def generate_text(model, seed, length=200, temperature=0.8):
    generated = seed
    for _ in range(length):
        x = np.array([[char2idx[c] for c in generated[-CHUNK:]]])
        probs = model.predict(x, verbose=0)[0] / temperature
        probs = np.exp(probs) / np.exp(probs).sum()
        next_idx = np.random.choice(len(probs), p=probs)
        generated += idx2char[next_idx]
    return generated

# pick a random seed from corpus
seed = text[:CHUNK]
generated_abc = generate_text(model, seed, length=300, temperature=0.8)
print(generated_abc)

with open('generated.abc', 'w') as f:
    f.write(generated_abc)
print("Saved generated.abc")
